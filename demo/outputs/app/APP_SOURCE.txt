import React, { useEffect, useMemo, useReducer, useRef } from "react";
import { Navigate, Route, Routes, useLocation, useNavigate } from "react-router-dom";
import HomePage from "./pages/HomePage.jsx";
import SettingsPage from "./pages/SettingsPage.jsx";
import { loadTasks, saveTasks } from "./storage/taskStorage.js";
import { loadTheme, saveTheme, resolveTheme, THEME_KEY } from "./storage/themeStorage.js";
import Toast from "./components/Toast.jsx";
import { createId } from "./utils/id.js";

const TASKS_KEY = "new-todoapp.tasks.v1";

const initialState = {
  hydrated: false,
  tasks: [],
  filter: "all",
  searchQuery: "",
  theme: "system", // system | light | dark
  toast: null
};

function sortTasks(tasks) {
  const active = tasks
    .filter((t) => t.status === "active")
    .sort((a, b) => b.createdAt - a.createdAt);
  const completed = tasks
    .filter((t) => t.status === "completed")
    .sort((a, b) => b.createdAt - a.createdAt);
  return [...active, ...completed];
}

function reducer(state, action) {
  switch (action.type) {
    case "HYDRATE": {
      const tasks = Array.isArray(action.tasks) ? sortTasks(action.tasks) : [];
      return {
        ...state,
        hydrated: true,
        tasks,
        theme: action.theme ?? state.theme
      };
    }
    case "SET_FILTER":
      return { ...state, filter: action.filter };
    case "SET_SEARCH":
      return { ...state, searchQuery: action.query };
    case "ADD_TASK": {
      const now = Date.now();
      const newTask = {
        id: createId(),
        title: action.title,
        note: action.note || "",
        status: "active",
        createdAt: now,
        updatedAt: now
      };
      return { ...state, tasks: sortTasks([newTask, ...state.tasks]) };
    }
    case "UPDATE_TASK": {
      const now = Date.now();
      const tasks = state.tasks.map((t) => {
        if (t.id !== action.id) return t;
        return {
          ...t,
          title: action.title,
          note: action.note ?? "",
          status: action.status ?? t.status,
          updatedAt: now
        };
      });
      return { ...state, tasks: sortTasks(tasks) };
    }
    case "TOGGLE_TASK": {
      const now = Date.now();
      const tasks = state.tasks.map((t) => {
        if (t.id !== action.id) return t;
        const nextStatus = t.status === "active" ? "completed" : "active";
        return { ...t, status: nextStatus, updatedAt: now };
      });
      return { ...state, tasks: sortTasks(tasks) };
    }
    case "DELETE_TASK": {
      const tasks = state.tasks.filter((t) => t.id !== action.id);
      return { ...state, tasks };
    }
    case "RESTORE_TASK": {
      const tasks = sortTasks([action.task, ...state.tasks]);
      return { ...state, tasks };
    }
    case "SET_THEME":
      return { ...state, theme: action.theme };
    case "SHOW_TOAST":
      return { ...state, toast: action.toast };
    case "HIDE_TOAST":
      if (!state.toast) return state;
      if (action.id && state.toast.id !== action.id) return state;
      return { ...state, toast: null };
    case "CLEAR_ALL":
      return { ...state, tasks: [] };
    default:
      return state;
  }
}

function useApplyTheme(theme) {
  useEffect(() => {
    const resolved = resolveTheme(theme);
    const root = document.documentElement;
    root.dataset.theme = resolved;
    root.style.colorScheme = resolved;

    if (theme !== "system") return;

    const mq = window.matchMedia?.("(prefers-color-scheme: dark)");
    if (!mq) return;

    const handler = () => {
      const r = resolveTheme("system");
      root.dataset.theme = r;
      root.style.colorScheme = r;
    };

    mq.addEventListener?.("change", handler);
    return () => mq.removeEventListener?.("change", handler);
  }, [theme]);
}

export default function App() {
  const [state, dispatch] = useReducer(reducer, initialState);
  const toastTimerRef = useRef(null);

  useApplyTheme(state.theme);

  // Hydrate
  useEffect(() => {
    const tasks = loadTasks(TASKS_KEY);
    const theme = loadTheme(THEME_KEY) ?? "system";
    dispatch({ type: "HYDRATE", tasks, theme });
  }, []);

  // Persist tasks
  useEffect(() => {
    if (!state.hydrated) return;
    saveTasks(TASKS_KEY, state.tasks);
  }, [state.hydrated, state.tasks]);

  // Persist theme
  useEffect(() => {
    if (!state.hydrated) return;
    saveTheme(THEME_KEY, state.theme);
  }, [state.hydrated, state.theme]);

  // Toast auto-hide
  useEffect(() => {
    if (!state.toast) return;
    if (toastTimerRef.current) window.clearTimeout(toastTimerRef.current);
    toastTimerRef.current = window.setTimeout(() => {
      dispatch({ type: "HIDE_TOAST", id: state.toast?.id });
    }, state.toast.durationMs ?? 4000);

    return () => {
      if (toastTimerRef.current) window.clearTimeout(toastTimerRef.current);
    };
  }, [state.toast]);

  const api = useMemo(() => {
    const showToast = (toast) => {
      dispatch({
        type: "SHOW_TOAST",
        toast: {
          id: createId(),
          durationMs: 4200,
          ...toast
        }
      });
    };

    const completeWithUndo = (taskId) => {
      const prev = state.tasks.find((t) => t.id === taskId);
      if (!prev) return;
      dispatch({ type: "TOGGLE_TASK", id: taskId });

      const willBeCompleted = prev.status === "active";
      showToast({
        message: willBeCompleted ? "Đã hoàn thành" : "Đã hoàn tác hoàn thành",
        actionLabel: "Hoàn tác",
        onAction: () => dispatch({ type: "TOGGLE_TASK", id: taskId })
      });
    };

    const deleteWithUndo = (taskId) => {
      const prev = state.tasks.find((t) => t.id === taskId);
      if (!prev) return;
      dispatch({ type: "DELETE_TASK", id: taskId });
      showToast({
        message: "Đã xoá",
        actionLabel: "Hoàn tác",
        onAction: () => dispatch({ type: "RESTORE_TASK", task: prev })
      });
    };

    return {
      dispatch,
      showToast,
      completeWithUndo,
      deleteWithUndo
    };
  }, [state.tasks]);

  return (
    <div className="app">
      <Routes>
        <Route
          path="/"
          element={
            <HomePage
              hydrated={state.hydrated}
              tasks={state.tasks}
              filter={state.filter}
              searchQuery={state.searchQuery}
              onChangeFilter={(filter) => dispatch({ type: "SET_FILTER", filter })}
              onChangeSearch={(query) => dispatch({ type: "SET_SEARCH", query })}
              onAddTask={({ title, note }) => dispatch({ type: "ADD_TASK", title, note })}
              onUpdateTask={({ id, title, note, status }) =>
                dispatch({ type: "UPDATE_TASK", id, title, note, status })
              }
              onToggleTask={(id) => api.completeWithUndo(id)}
              onDeleteTask={(id) => api.deleteWithUndo(id)}
              onShowToast={api.showToast}
            />
          }
        />
        <Route
          path="/settings"
          element={
            <SettingsPage
              theme={state.theme}
              onChangeTheme={(theme) => dispatch({ type: "SET_THEME", theme })}
              onClearAll={() => dispatch({ type: "CLEAR_ALL" })}
              onShowToast={api.showToast}
            />
          }
        />
        <Route path="*" element={<Navigate to="/" replace />} />
      </Routes>

      <Toast
        toast={state.toast}
        onClose={() => dispatch({ type: "HIDE_TOAST", id: state.toast?.id })}
      />
    </div>
  );
}

export function loadTasks(storageKey) {
  try {
    const raw = localStorage.getItem(storageKey);
    if (!raw) return [];
    const parsed = JSON.parse(raw);
    if (!Array.isArray(parsed)) return [];
    // basic shape cleanup
    return parsed
      .filter((t) => t && typeof t.id === "string" && typeof t.title === "string")
      .map((t) => ({
        id: String(t.id),
        title: String(t.title),
        note: typeof t.note === "string" ? t.note : "",
        status: t.status === "completed" ? "completed" : "active",
        createdAt: typeof t.createdAt === "number" ? t.createdAt : Date.now(),
        updatedAt: typeof t.updatedAt === "number" ? t.updatedAt : Date.now()
      }));
  } catch {
    return [];
  }
}

export function saveTasks(storageKey, tasks) {
  try {
    localStorage.setItem(storageKey, JSON.stringify(tasks ?? []));
  } catch {
    // ignore quota / private mode errors
  }
}

export const THEME_KEY = "new-todoapp.theme.v1";

export function loadTheme(key) {
  try {
    const v = localStorage.getItem(key);
    if (v === "light" || v === "dark" || v === "system") return v;
    return null;
  } catch {
    return null;
  }
}

export function saveTheme(key, theme) {
  try {
    localStorage.setItem(key, theme);
  } catch {
    // ignore
  }
}

export function resolveTheme(theme) {
  if (theme === "light" || theme === "dark") return theme;
  const mq = window.matchMedia?.("(prefers-color-scheme: dark)");
  const isDark = mq ? mq.matches : false;
  return isDark ? "dark" : "light";
}

export function createId() {
  // Prefer crypto.randomUUID when available
  if (typeof crypto !== "undefined" && crypto.randomUUID) return crypto.randomUUID();
  return `id_${Math.random().toString(16).slice(2)}_${Date.now().toString(16)}`;
}

import { useEffect, useState } from "react";

export function useDebouncedValue(value, delayMs = 200) {
  const [debounced, setDebounced] = useState(value);

  useEffect(() => {
    const t = window.setTimeout(() => setDebounced(value), delayMs);
    return () => window.clearTimeout(t);
  }, [value, delayMs]);

  return debounced;
}

import React, { useEffect, useMemo, useRef, useState } from "react";
import ModalSheet from "./ModalSheet.jsx";

const MAX_TITLE = 120;
const MAX_NOTE = 500;

export default function TaskModal({
  open,
  mode, // add | edit
  task,
  onClose,
  onSubmit,
  onRequestDelete
}) {
  const titleText = mode === "add" ? "Thêm việc" : "Sửa việc";

  const [title, setTitle] = useState("");
  const [note, setNote] = useState("");
  const [completed, setCompleted] = useState(false);
  const [error, setError] = useState("");

  const titleInputRef = useRef(null);

  useEffect(() => {
    if (!open) return;
    if (mode === "edit" && task) {
      setTitle(task.title ?? "");
      setNote(task.note ?? "");
      setCompleted(task.status === "completed");
    } else {
      setTitle("");
      setNote("");
      setCompleted(false);
    }
    setError("");
  }, [open, mode, task]);

  const canDelete = mode === "edit" && task;

  const handleSubmit = () => {
    const trimmed = (title || "").trim();
    if (!trimmed) {
      setError("Vui lòng nhập tiêu đề");
      titleInputRef.current?.focus?.();
      return;
    }
    if (trimmed.length > MAX_TITLE) {
      setError(`Tiêu đề tối đa ${MAX_TITLE} ký tự`);
      titleInputRef.current?.focus?.();
      return;
    }
    const safeNote = (note || "").slice(0, MAX_NOTE);

    onSubmit({
      title: trimmed,
      note: safeNote,
      status: mode === "edit" ? (completed ? "completed" : "active") : "active"
    });
  };

  const footer = useMemo(() => {
    return (
      <div className="modalFooter">
        {canDelete ? (
          <button type="button" className="button button--danger button--ghost" onClick={onRequestDelete}>
            Xoá
          </button>
        ) : (
          <span />
        )}
        <div className="modalFooter__right">
          <button type="button" className="button button--ghost" onClick={onClose}>
            Huỷ
          </button>
          <button type="button" className="button button--primary" onClick={handleSubmit}>
            Lưu
          </button>
        </div>
      </div>
    );
  }, [canDelete, onClose, onRequestDelete, handleSubmit]);

  return (
    <ModalSheet
      open={open}
      title={titleText}
      onClose={onClose}
      initialFocusRef={titleInputRef}
      footer={footer}
    >
      <div className="form">
        <label className="label" htmlFor="taskTitle">
          Tiêu đề <span className="required">*</span>
        </label>
        <input
          id="taskTitle"
          ref={titleInputRef}
          className={`input ${error ? "input--error" : ""}`}
          value={title}
          onChange={(e) => {
            setTitle(e.target.value);
            if (error) setError("");
          }}
          placeholder="Ví dụ: Gọi điện cho khách hàng"
          maxLength={MAX_TITLE + 20}
          autoComplete="off"
          autoCorrect="off"
          spellCheck={false}
        />
        {error ? <div className="errorText" role="alert">{error}</div> : null}

        <label className="label" htmlFor="taskNote">
          Ghi chú
        </label>
        <textarea
          id="taskNote"
          className="textarea"
          value={note}
          onChange={(e) => setNote(e.target.value)}
          placeholder="(Tuỳ chọn)"
          maxLength={MAX_NOTE + 50}
          rows={4}
        />

        {mode === "edit" ? (
          <div className="row">
            <label className="toggle">
              <input
                type="checkbox"
                checked={completed}
                onChange={(e) => setCompleted(e.target.checked)}
                aria-label="Hoàn thành"
              />
              <span>Hoàn thành</span>
            </label>
          </div>
        ) : null}

        <div className="hint">
          Tiêu đề tối đa {MAX_TITLE} ký tự. Ghi chú tối đa {MAX_NOTE} ký tự.
        </div>
      </div>
    </ModalSheet>
  );
}

import React, { useEffect, useMemo, useRef } from "react";

function useIsDesktop() {
  const mq = typeof window !== "undefined" ? window.matchMedia?.("(min-width: 640px)") : null;
  const [isDesktop, setIsDesktop] = React.useState(() => (mq ? mq.matches : false));

  useEffect(() => {
    if (!mq) return;
    const handler = () => setIsDesktop(mq.matches);
    mq.addEventListener?.("change", handler);
    return () => mq.removeEventListener?.("change", handler);
  }, [mq]);

  return isDesktop;
}

export default function ModalSheet({ open, title, children, onClose, footer, initialFocusRef }) {
  const isDesktop = useIsDesktop();
  const dialogRef = useRef(null);
  const contentRef = useRef(null);
  const lastActiveElRef = useRef(null);

  useEffect(() => {
    if (!open) return;

    lastActiveElRef.current = document.activeElement;

    const onKeyDown = (e) => {
      if (e.key === "Escape") onClose?.();
      if (e.key !== "Tab") return;

      const root = contentRef.current;
      if (!root) return;

      const focusables = root.querySelectorAll(
        'a[href], button:not([disabled]), textarea:not([disabled]), input:not([disabled]), select:not([disabled]), [tabindex]:not([tabindex="-1"])'
      );
      if (!focusables.length) return;

      const first = focusables[0];
      const last = focusables[focusables.length - 1];

      if (e.shiftKey && document.activeElement === first) {
        e.preventDefault();
        last.focus();
      } else if (!e.shiftKey && document.activeElement === last) {
        e.preventDefault();
        first.focus();
      }
    };

    document.addEventListener("keydown", onKeyDown);

    // prevent background scroll
    const prevOverflow = document.body.style.overflow;
    document.body.style.overflow = "hidden";

    const focusTarget = initialFocusRef?.current;
    window.setTimeout(() => focusTarget?.focus?.(), 0);

    return () => {
      document.removeEventListener("keydown", onKeyDown);
      document.body.style.overflow = prevOverflow;
      const last = lastActiveElRef.current;
      if (last && typeof last.focus === "function") last.focus();
    };
  }, [open, onClose, initialFocusRef]);

  const containerClass = useMemo(() => {
    if (!open) return "modal modal--closed";
    return `modal ${isDesktop ? "modal--dialog" : "modal--sheet"}`;
  }, [open, isDesktop]);

  if (!open) return null;

  return (
    <div className={containerClass} role="presentation">
      <div className="modal__backdrop" onClick={onClose} aria-hidden="true" />
      <section
        className="modal__panel"
        role="dialog"
        aria-modal="true"
        aria-label={title}
        ref={dialogRef}
      >
        <div className="modal__handle" aria-hidden="true" />
        <div className="modal__header">
          <div className="modal__title">{title}</div>
          <button className="iconButton" type="button" onClick={onClose} aria-label="Đóng">
            ✕
          </button>
        </div>

        <div className="modal__content" ref={contentRef}>
          {children}
        </div>

        {footer ? <div className="modal__footer">{footer}</div> : null}
      </section>
    </div>
  );
}

function matchesQuery(title, q) {
  const query = (q || "").trim().toLowerCase();
  if (!query) return true;
  return (title || "").toLowerCase().includes(query);
}

function filterTasks(tasks, filter, q) {
  const query = (q || "").trim();
  const byStatus =
    filter === "all" ? tasks : tasks.filter((t) => (filter === "active" ? t.status === "active" : t.status === "completed"));
  if (!query) return byStatus;
  return byStatus.filter((t) => matchesQuery(t.title, query));
}

export default function HomePage({
  hydrated,
  tasks,
  filter,
  searchQuery,
  onChangeFilter,
  onChangeSearch,
  onAddTask,
  onUpdateTask,
  onToggleTask,
  onDeleteTask,
  onShowToast
}) {
  const [isAddOpen, setIsAddOpen] = useState(false);
  const [editingTaskId, setEditingTaskId] = useState(null);
  const [confirmDeleteTaskId, setConfirmDeleteTaskId] = useState(null);
  const [actionTaskId, setActionTaskId] = useState(null);

  const debouncedQuery = useDebouncedValue(searchQuery, 200);

  const visibleTasks = useMemo(() => {
    return filterTasks(tasks, filter, debouncedQuery);
  }, [tasks, filter, debouncedQuery]);

  const editingTask = useMemo(() => {
    if (!editingTaskId) return null;
    return tasks.find((t) => t.id === editingTaskId) ?? null;
  }, [tasks, editingTaskId]);

  const actionTask = useMemo(() => {
    if (!actionTaskId) return null;
    return tasks.find((t) => t.id === actionTaskId) ?? null;
  }, [tasks, actionTaskId]);

  const emptyText = useMemo(() => {
    const q = (debouncedQuery || "").trim();
    if (q) return "Không tìm thấy việc phù hợp.";
    if (filter === "completed") return "Chưa có việc hoàn thành.";
    if (filter === "active") return "Chưa có việc đang làm.";
    return "Chưa có việc nào. Thêm việc đầu tiên.";
  }, [filter, debouncedQuery]);

  return (
    <div className="page">
      <AppHeader title="Todo" />

      <main className="content">
        <div className="topControls">
          <SearchBar value={searchQuery} onChange={onChangeSearch} />
          <FilterChips value={filter} onChange={onChangeFilter} />
        </div>

        {!hydrated ? (
          <div className="loading" aria-label="Đang tải">
            <div className="spinner" aria-hidden="true" />
            <div className="muted">Đang tải...</div>
          </div>
        ) : visibleTasks.length > 0 ? (
          <TaskList
            tasks={visibleTasks}
            onToggle={onToggleTask}
            onOpenEdit={(id) => setEditingTaskId(id)}
            onOpenActions={(id) => setActionTaskId(id)}
          />
        ) : (
          <div className="empty" role="status" aria-live="polite">
            <div className="empty__card">
              <div className="empty__title">{emptyText}</div>
              <button type="button" className="button button--primary button--block" onClick={() => setIsAddOpen(true)}>
                Thêm việc
              </button>
            </div>
          </div>
        )}
      </main>

      <FabAddButton onClick={() => setIsAddOpen(true)} />

      <TaskModal
        open={isAddOpen}
        mode="add"
        task={null}
        onClose={() => setIsAddOpen(false)}
        onSubmit={({ title, note }) => {
          onAddTask({ title, note });
          setIsAddOpen(false);
          onShowToast?.({ message: "Đã thêm việc" });
        }}
      />

      <TaskModal
        open={Boolean(editingTaskId && editingTask)}
        mode="edit"
        task={editingTask}
        onClose={() => setEditingTaskId(null)}
        onRequestDelete={() => {
          if (!editingTaskId) return;
          setConfirmDeleteTaskId(editingTaskId);
        }}
        onSubmit={({ title, note, status }) => {
          if (!editingTaskId) return;
          onUpdateTask({ id: editingTaskId, title, note, status });
          setEditingTaskId(null);
          onShowToast?.({ message: "Đã lưu thay đổi" });
        }}
      />

      <ConfirmDialog
        open={Boolean(confirmDeleteTaskId)}
        title="Xoá việc?"
        description="Hành động này sẽ xoá việc khỏi danh sách. Bạn có thể hoàn tác ngay sau đó."
        confirmLabel="Xoá"
        danger
        onClose={() => setConfirmDeleteTaskId(null)}
        onConfirm={() => {
          const id = confirmDeleteTaskId;
          setConfirmDeleteTaskId(null);
          setEditingTaskId(null);
          setActionTaskId(null);
          if (!id) return;
          onDeleteTask(id);
        }}
      />

      <ActionSheet
        open={Boolean(actionTaskId && actionTask)}
        title="Tùy chọn"
        onClose={() => setActionTaskId(null)}
        actions={[
          {
            key: "edit",
            label: "Sửa",
            onClick: () => {
              const id = actionTaskId;
              setActionTaskId(null);
              if (id) setEditingTaskId(id);
            }
          },
          {
            key: "delete",
            label: "Xoá",
            danger: true,
            onClick: () => {
              const id = actionTaskId;
              setActionTaskId(null);
              if (id) setConfirmDeleteTaskId(id);
            }
          }
        ]}
      />
    </div>
  );
}
